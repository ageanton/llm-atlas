```ts
import type { AblationConfig } from '../engine/types';

export type Register = 'simple' | 'sufficient' | 'complete';

export interface AtlasState {
  prompt: string;
  nodeId: string;
  layer: number;
  head: number;
  register: Register;
  ablation: AblationConfig;
  showPreSoftmax: boolean;
  paramName: string;
  paramRow: number;
  paramCol: number;
}

export const DEFAULT_PROMPT =
  'Once upon a time, a little dog named Max found a red ball. He';

export function clampIndex(n: number, max = 3): number {
  if (!Number.isFinite(n) || n < 0) return 0;
  if (n > max) return max;
  return Math.round(n);
}

export function hasActiveAblation(ablation: AblationConfig): boolean {
  return (
    (ablation.disabledHeads?.length ?? 0) > 0 ||
    (ablation.disabledAttnBlocks?.length ?? 0) > 0 ||
    (ablation.disabledMlpBlocks?.length ?? 0) > 0 ||
    (ablation.disabledResiduals?.length ?? 0) > 0 ||
    (ablation.disabledLayerNorms?.length ?? 0) > 0 ||
    (ablation.disabledLayers?.length ?? 0) > 0
  );
}

export function isHeadDisabled(
  ablation: AblationConfig,
  layer: number,
  head: number
): boolean {
  return (ablation.disabledHeads ?? []).some(
    ([l, h]) => l === layer && h === head
  );
}

export function ablationToParam(ablation: AblationConfig): string {
  const parts: string[] = [];
  for (const [l, h] of ablation.disabledHeads ?? []) parts.push(`h${l}_${h}`);
  for (const l of ablation.disabledAttnBlocks ?? []) parts.push(`a${l}`);
  for (const l of ablation.disabledMlpBlocks ?? []) parts.push(`m${l}`);
  for (const [kind, l] of ablation.disabledResiduals ?? [])
    parts.push(`r${kind}_${l}`);
  for (const [kind, l] of ablation.disabledLayerNorms ?? [])
    parts.push(`n${kind}_${l}`);
  for (const l of ablation.disabledLayers ?? []) parts.push(`l${l}`);
  return parts.length > 0 ? parts.join(',') : 'none';
}

export function paramToAblation(raw: string | null): AblationConfig {
  const cfg: AblationConfig = {};
  if (!raw || raw === 'none') return cfg;
  for (const token of raw.split(',')) {
    if (token.startsWith('h')) {
      const [l, h] = token.slice(1).split('_').map(Number);
      if (Number.isFinite(l) && Number.isFinite(h)) {
        cfg.disabledHeads = [
          ...(cfg.disabledHeads ?? []),
          [clampIndex(l), clampIndex(h)],
        ];
      }
    } else if (token.startsWith('a')) {
      const l = Number(token.slice(1));
      if (Number.isFinite(l)) {
        cfg.disabledAttnBlocks = [
          ...(cfg.disabledAttnBlocks ?? []),
          clampIndex(l),
        ];
      }
    } else if (token.startsWith('m')) {
      const l = Number(token.slice(1));
      if (Number.isFinite(l)) {
        cfg.disabledMlpBlocks = [
          ...(cfg.disabledMlpBlocks ?? []),
          clampIndex(l),
        ];
      }
    } else if (token.startsWith('r')) {
      const [kind, lStr] = token.slice(1).split('_');
      const l = Number(lStr);
      if ((kind === 'attn' || kind === 'mlp') && Number.isFinite(l)) {
        cfg.disabledResiduals = [
          ...(cfg.disabledResiduals ?? []),
          [kind, clampIndex(l)],
        ];
      }
    } else if (token.startsWith('n')) {
      const idx = token.lastIndexOf('_');
      const kind = token.slice(1, idx);
      const l = Number(token.slice(idx + 1));
      if (
        (kind === 'ln_1' || kind === 'ln_2' || kind === 'ln_f') &&
        Number.isFinite(l)
      ) {
        cfg.disabledLayerNorms = [
          ...(cfg.disabledLayerNorms ?? []),
          [kind, kind === 'ln_f' ? -1 : clampIndex(l)],
        ];
      }
    } else if (token.startsWith('l')) {
      const l = Number(token.slice(1));
      if (Number.isFinite(l)) {
        cfg.disabledLayers = [...(cfg.disabledLayers ?? []), clampIndex(l)];
      }
    }
  }
  return cfg;
}

function parseRegister(raw: string | null): Register {
  if (raw === 'simple' || raw === 'simp' || raw === 'intuitive') return 'simple';
  if (raw === 'complete' || raw === 'comp' || raw === 'math') return 'complete';
  return 'sufficient';
}

export function stateFromUrl(search: string): AtlasState {
  const params = new URLSearchParams(search);
  const prompt = params.get('in') || DEFAULT_PROMPT;
  const nodeId = params.get('node') || 'root';
  const layer = clampIndex(Number(params.get('block') ?? '2'));
  const head = clampIndex(Number(params.get('head') ?? '3'));
  const register = parseRegister(params.get('reg'));
  const ablation = paramToAblation(params.get('abl'));
  const showPreSoftmax = params.get('raw') === '1';
  const paramName = params.get('param') || 'blocks.2.attn.W_k.weight';
  const paramRow = Math.max(0, Number(params.get('r') ?? '64') || 0);
  const paramCol = Math.max(0, Number(params.get('c') ?? '19') || 0);

  return {
    prompt,
    nodeId,
    layer,
    head,
    register,
    ablation,
    showPreSoftmax,
    paramName,
    paramRow,
    paramCol,
  };
}

export function stateToSearch(state: AtlasState): string {
  const params = new URLSearchParams();
  params.set('in', state.prompt);
  params.set('node', state.nodeId);
  params.set('block', String(state.layer));
  params.set('head', String(state.head));
  params.set('reg', state.register);
  params.set('abl', ablationToParam(state.ablation));
  if (state.showPreSoftmax) params.set('raw', '1');
  if (state.nodeId === 'scalar') {
    params.set('param', state.paramName);
    params.set('r', String(state.paramRow));
    params.set('c', String(state.paramCol));
  }
  return `?${params.toString()}`;
}
```

```ts
import * as THREE from 'three';

export function sequentialColor(tRaw: number): string {
  const t = Math.max(0, Math.min(1, tRaw));
  const stops: [number, [number, number, number]][] = [
    [0.0, [242, 245, 250]],
    [0.25, [191, 219, 254]],
    [0.55, [59, 130, 246]],
    [0.82, [29, 78, 216]],
    [1.0, [23, 37, 84]],
  ];
  return interpolateStops(t, stops);
}

export function divergingColor(normRaw: number): string {
  const v = Math.max(-1, Math.min(1, normRaw));
  const t = (v + 1) / 2;
  const stops: [number, [number, number, number]][] = [
    [0.0, [30, 58, 138]],
    [0.25, [96, 165, 250]],
    [0.5, [248, 250, 252]],
    [0.75, [251, 146, 60]],
    [1.0, [180, 83, 9]],
  ];
  return interpolateStops(t, stops);
}

function interpolateStops(
  t: number,
  stops: [number, [number, number, number]][]
): string {
  for (let i = 0; i < stops.length - 1; i++) {
    const [t0, c0] = stops[i];
    const [t1, c1] = stops[i + 1];
    if (t >= t0 && t <= t1) {
      const f = t1 === t0 ? 0 : (t - t0) / (t1 - t0);
      const r = Math.round(c0[0] + (c1[0] - c0[0]) * f);
      const g = Math.round(c0[1] + (c1[1] - c0[1]) * f);
      const b = Math.round(c0[2] + (c1[2] - c0[2]) * f);
      return `rgb(${r}, ${g}, ${b})`;
    }
  }
  const last = stops[stops.length - 1][1];
  return `rgb(${last[0]}, ${last[1]}, ${last[2]})`;
}

export function formatToken(raw: string): string {
  if (raw === '\n') return '↵';
  if (raw === ' ') return '␣';
  return raw.replace(/^ /, '');
}

export function formatTokenExplicit(raw: string): string {
  if (raw === '\n') return '\\n';
  if (raw === ' ') return '·';
  return raw.replace(/^ /, '·');
}

export function formatNum(n: number, digits = 3): string {
  if (!Number.isFinite(n)) return '0';
  return n.toFixed(digits);
}

export function vectorL2Norm(vec: number[]): number {
  let sum = 0;
  for (let i = 0; i < vec.length; i++) sum += vec[i] * vec[i];
  return Math.sqrt(sum);
}

export function countActiveNeurons(vec: number[], threshold = 0.05): number {
  let count = 0;
  for (let i = 0; i < vec.length; i++) {
    if (vec[i] > threshold) count++;
  }
  return count;
}

export interface SampledToken {
  id: number;
  prob: number;
  logit: number;
}

export function computeSampledDistribution(
  logits: number[],
  temperature: number,
  topK: number,
  topP: number
): SampledToken[] {
  const temp = Math.max(0.05, temperature);
  const scaled = logits.map((v) => v / temp);

  let maxVal = -Infinity;
  for (let i = 0; i < scaled.length; i++) {
    if (scaled[i] > maxVal) maxVal = scaled[i];
  }

  const indexed = scaled.map((v, id) => ({
    id,
    logit: logits[id],
    exp: Math.exp(v - maxVal),
  }));
  indexed.sort((a, b) => b.exp - a.exp || a.id - b.id);

  const kClamped = Math.max(1, Math.min(indexed.length, Math.round(topK)));
  const afterK = indexed.slice(0, kClamped);

  let kSum = 0;
  for (const item of afterK) kSum += item.exp;

  const afterP: SampledToken[] = [];
  let cumulative = 0;
  for (const item of afterK) {
    const p = item.exp / kSum;
    afterP.push({ id: item.id, prob: p, logit: item.logit });
    cumulative += p;
    if (cumulative >= topP && afterP.length >= 1) break;
  }

  let finalSum = 0;
  for (const item of afterP) finalSum += item.prob;
  return afterP.map((item) => ({
    id: item.id,
    logit: item.logit,
    prob: finalSum > 0 ? item.prob / finalSum : 0,
  }));
}

export function sampleFromDistribution(dist: SampledToken[]): number {
  const r = Math.random();
  let acc = 0;
  for (const item of dist) {
    acc += item.prob;
    if (r <= acc) return item.id;
  }
  return dist[dist.length - 1]?.id ?? 0;
}

export function createTensorTexture(
  matrix: number[][],
  mode: 'diverging' | 'sequential' = 'diverging',
  transpose = false
): THREE.CanvasTexture {
  const srcRows = matrix.length;
  const srcCols = matrix[0]?.length ?? 1;
  const rows = transpose ? srcCols : srcRows;
  const cols = transpose ? srcRows : srcCols;

  const canvas = document.createElement('canvas');
  canvas.width = Math.max(1, Math.min(cols, 512));
  canvas.height = Math.max(1, Math.min(rows, 512));
  const ctx = canvas.getContext('2d')!;

  let maxAbs = 1e-6;
  let minVal = Infinity;
  let maxVal = -Infinity;
  for (let r = 0; r < srcRows; r++) {
    const row = matrix[r];
    for (let c = 0; c < srcCols; c++) {
      const v = row[c];
      if (v < minVal) minVal = v;
      if (v > maxVal) maxVal = v;
      const a = Math.abs(v);
      if (a > maxAbs) maxAbs = a;
    }
  }
  if (!Number.isFinite(minVal)) minVal = 0;
  if (!Number.isFinite(maxVal)) maxVal = 1;

  const w = canvas.width;
  const h = canvas.height;
  for (let py = 0; py < h; py++) {
    const r = Math.min(rows - 1, Math.floor((py / h) * rows));
    for (let px = 0; px < w; px++) {
      const c = Math.min(cols - 1, Math.floor((px / w) * cols));
      const val = transpose ? matrix[c][r] : matrix[r][c];
      if (mode === 'sequential') {
        const norm =
          maxVal > minVal ? (val - minVal) / (maxVal - minVal) : 0;
        ctx.fillStyle = sequentialColor(norm);
      } else {
        ctx.fillStyle = divergingColor(val / maxAbs);
      }
      ctx.fillRect(px, py, 1, 1);
    }
  }

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.magFilter = THREE.NearestFilter;
  tex.minFilter = THREE.LinearFilter;
  tex.generateMipmaps = false;
  return tex;
}

function roundRect(
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  w: number,
  h: number,
  r: number
): void {
  const rad = Math.min(r, w / 2, h / 2);
  ctx.beginPath();
  ctx.moveTo(x + rad, y);
  ctx.arcTo(x + w, y, x + w, y + h, rad);
  ctx.arcTo(x + w, y + h, x, y + h, rad);
  ctx.arcTo(x, y + h, x, y, rad);
  ctx.arcTo(x, y, x + w, y, rad);
  ctx.closePath();
}

export function createPlaqueTexture(
  title: string,
  subtitle?: string,
  accent = '#1d4ed8',
  muted = false
): THREE.CanvasTexture {
  const canvas = document.createElement('canvas');
  canvas.width = 420;
  canvas.height = subtitle ? 124 : 84;
  const ctx = canvas.getContext('2d')!;

  ctx.clearRect(0, 0, canvas.width, canvas.height);
  roundRect(ctx, 8, 8, canvas.width - 16, canvas.height - 16, 18);
  ctx.fillStyle = muted
    ? 'rgba(241, 245, 249, 0.94)'
    : 'rgba(255, 255, 255, 0.95)';
  ctx.fill();
  ctx.lineWidth = 3;
  ctx.strokeStyle = muted ? '#cbd5e1' : accent;
  ctx.stroke();

  ctx.fillStyle = muted ? '#64748b' : '#141923';
  ctx.font =
    '700 28px -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(title.slice(0, 24), canvas.width / 2, subtitle ? 44 : 42);

  if (subtitle) {
    ctx.fillStyle = muted ? '#b45309' : accent;
    ctx.font = '600 21px ui-monospace, SFMono-Regular, Menlo, monospace';
    ctx.fillText(subtitle.slice(0, 30), canvas.width / 2, 86);
  }

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  return tex;
}
```

```ts
  const chain: AnatomyNode[] = [];
  let cur: AnatomyNode | undefined = getAnatomyNode(nodeId);
  const seen = new Set<string>();
  while (cur && !seen.has(cur.id)) {
    seen.add(cur.id);
    chain.unshift(cur);
    cur = cur.parentId ? ANATOMY_NODES[cur.parentId] : undefined;
  }
  return chain;
```

```ts
  private pickObject(event: PointerEvent | MouseEvent): ClickTag | null {
    const rect = this.renderer.domElement.getBoundingClientRect();
    this.pointer.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
    this.pointer.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
    this.raycaster.setFromCamera(this.pointer, this.camera);
    const hits = this.raycaster.intersectObjects(this.interactives, false);
    for (const hit of hits) {
      const tag = hit.object.userData as ClickTag | undefined;
      if (tag?.kind) return tag;
    }
    return null;
  }

  private readonly onPointerMove = (event: PointerEvent): void => {
    const hit = this.pickObject(event);
    this.renderer.domElement.style.cursor = hit ? 'pointer' : 'grab';
    if (hit?.kind === 'token' && typeof hit.tokenIdx === 'number') {
      if (this.lastHoverToken !== hit.tokenIdx) {
        this.lastHoverToken = hit.tokenIdx;
        this.callbacks.onHoverToken(hit.tokenIdx);
      }
    } else if (this.lastHoverToken !== null) {
      this.lastHoverToken = null;
      this.callbacks.onHoverToken(null);
    }
  };

  private readonly onClick = (event: MouseEvent): void => {
    const hit = this.pickObject(event);
    if (!hit) return;
    if (hit.kind === 'node' && hit.nodeId) {
      this.callbacks.onSelectNode(hit.nodeId, hit.layer);
    }
  };

  private readonly onResize = (): void => {
    if (this.disposed) return;
    const w = Math.max(1, this.container.clientWidth);
    const h = Math.max(1, this.container.clientHeight);
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(w, h);
  };

  private readonly animate = (): void => {
    if (this.disposed) return;
    this.animFrameId = requestAnimationFrame(this.animate);

    const dt = Math.min(0.05, this.clock.getDelta());
    const t = this.clock.elapsedTime;

    if (this.needsCameraFlight && !this.isDragging) {
      const ease = 1 - Math.exp(-5.5 * dt);
      this.camera.position.lerp(this.camTargetPos, ease);
      this.controls.target.lerp(this.camTargetLook, ease);
      if (
        this.camera.position.distanceTo(this.camTargetPos) < 0.03 &&
        this.controls.target.distanceTo(this.camTargetLook) < 0.03
      ) {
        this.needsCameraFlight = false;
      }
    }
    this.controls.update();

    if (this.flowBead) {
      const frac = (t * 0.26) % 1;
      this.flowBead.position.x = -5.4 + frac * 11.4;
    }

    this.renderer.render(this.scene, this.camera);
  };

  dispose(): void {
    this.disposed = true;
    cancelAnimationFrame(this.animFrameId);
    window.removeEventListener('resize', this.onResize);
    this.renderer.domElement.removeEventListener('pointermove', this.onPointerMove);
    this.renderer.domElement.removeEventListener('click', this.onClick);
    this.clearStage();
    this.controls.dispose();
    this.envRT.dispose();
    this.renderer.dispose();
    if (this.renderer.domElement.parentNode === this.container) {
      this.container.removeChild(this.renderer.domElement);
    }
  }
}
```

```tsx
  const lt = res.trace.layers[layer];
  if (lt.skipped) {
    return (
      <div className="lab-card muted-sm">
        Block {layer} is currently skipped by Ablation Lab.
      </div>
    );
  }
  const active: ActiveLayerTrace = lt;
  const mat = showPreSoftmax
    ? active.attn.rawScores[head]
    : active.attn.attnWeights[head];
  const tokens = res.tokens;

  let maxAbs = 1e-6;
  for (let r = 0; r < tokens.length; r++) {
    for (let c = 0; c <= r; c++) {
      maxAbs = Math.max(maxAbs, Math.abs(active.attn.rawScores[head][r][c]));
    }
  }

  const focusRow = hoveredTokenIdx ?? tokens.length - 1;
  const rowSum = active.attn.attnWeights[head][focusRow].reduce((a, b) => a + b, 0);

  return (
    <div className="lab-card">
      <div className="row-between" style={{ marginBottom: 8 }}>
        <span className="lab-card-title" style={{ margin: 0 }}>
          Block {layer} · Head {head} [{tokens.length}×{tokens.length}]
        </span>
        <button
          type="button"
          className={`mini-btn ${showPreSoftmax ? 'is-active' : ''}`}
          onClick={onTogglePreSoftmax}
        >
          {showPreSoftmax ? 'Raw QKᵀ/√32' : 'Softmax A'}
        </button>
      </div>
      <div className="matrix-scroll">
        <table className="exact-table mono">
          <thead>
            <tr>
              <th>Q\K</th>
              {tokens.map((t, c) => (
                <th
                  key={`c-${c}`}
                  className={hoveredTokenIdx === c ? 'is-cross' : ''}
                  onMouseEnter={() => onHoverToken(c)}
                  onMouseLeave={() => onHoverToken(null)}
                >
                  {formatTokenExplicit(t.text)}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {tokens.map((rowTok, r) => (
              <tr key={`r-${r}`}>
                <th
                  className={hoveredTokenIdx === r ? 'is-cross' : ''}
                  onMouseEnter={() => onHoverToken(r)}
                  onMouseLeave={() => onHoverToken(null)}
                >
                  {formatTokenExplicit(rowTok.text)}
                </th>
                {tokens.map((_, c) => {
                  const masked = c > r;
                  const v = mat[r][c];
                  const bg = masked
                    ? '#f1f5f9'
                    : showPreSoftmax
                      ? divergingColor(v / maxAbs)
                      : sequentialColor(v);
                  const dark = !masked && !showPreSoftmax && v > 0.45;
                  return (
                    <td
                      key={`td-${r}-${c}`}
                      style={{ backgroundColor: bg, color: dark ? '#fff' : '#141923' }}
                      onMouseEnter={() => !masked && onHoverToken(r)}
                      onMouseLeave={() => onHoverToken(null)}
                    >
                      {masked ? '' : showPreSoftmax ? formatNum(v, 1) : (v * 100).toFixed(0)}
                    </td>
                  );
                })}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <div className="muted-xs mono" style={{ marginTop: 6 }}>
        Row “{formatTokenExplicit(tokens[focusRow].text)}” post-softmax sum = {rowSum.toFixed(6)}
      </div>
    </div>
  );
}

function InlineWeightMatrix({
  client,
  manifest,
  paramName,
  onPickScalar,
}: {
  client: TransformerEngineClient;
  manifest: ModelManifest;
  paramName: string;
  onPickScalar: (param: string, row: number, col: number) => void;
}) {
  const canvasRef = useRef<HTMLCanvasElement null |>(null);
  const [data, setData] = useState<Float32Array null |>(null);
  const [hover, setHover] = useState<{ r: number; c: number; v: number } | null>(null);

  const meta =
    manifest.tensors.find((t) => t.name === paramName) ?? manifest.tensors[0];
  const is1D = meta.shape.length === 1;
  const rows = meta.shape[0];
  const cols = is1D ? 1 : (meta.shape[1] ?? 1);

  useEffect(() => {
    let alive = true;
    client.getTensor(meta.name).then((arr) => {
      if (alive) setData(arr);
    });
    return () => {
      alive = false;
    };
  }, [client, meta.name]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas || !data) return;
    canvas.width = is1D ? rows : cols;
    canvas.height = is1D ? 1 : rows;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const maxAbs = Math.max(Math.abs(meta.stats.min), Math.abs(meta.stats.max), 1e-6);
    if (is1D) {
      for (let i = 0; i < rows; i++) {
        ctx.fillStyle = divergingColor((data[i] ?? 0) / maxAbs);
        ctx.fillRect(i, 0, 1, 1);
      }
    } else {
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          ctx.fillStyle = divergingColor((data[r * cols + c] ?? 0) / maxAbs);
          ctx.fillRect(c, r, 1, 1);
        }
      }
    }
  }, [data, rows, cols, is1D, meta.stats.min, meta.stats.max]);

  const pick = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!data) return null;
    const rect = e.currentTarget.getBoundingClientRect();
    const x = Math.max(0, Math.min(rect.width - 0.001, e.clientX - rect.left));
    const y = Math.max(0, Math.min(rect.height - 0.001, e.clientY - rect.top));
    if (is1D) {
      const r = Math.min(rows - 1, Math.floor((x / rect.width) * rows));
      return { r, c: 0, v: data[r] ?? 0 };
    }
    const c = Math.min(cols - 1, Math.floor((x / rect.width) * cols));
    const r = Math.min(rows - 1, Math.floor((y / rect.height) * rows));
    return { r, c, v: data[r * cols + c] ?? 0 };
  };

  return (
    <div className="lab-card">
      <div className="row-between" style={{ marginBottom: 6 }}>
        <span className="lab-card-title mono" style={{ margin: 0 }}>
          {meta.name} [{meta.shape.join('×')}]
        </span>
        <span className="muted-xs mono">{meta.count.toLocaleString()} params</span>
      </div>
      <canvas
        ref={canvasRef}
        className="weight-canvas"
        style={{ height: is1D ? 44 : 185 }}
        onMouseMove={(e) => setHover(pick(e))}
        onMouseLeave={() => setHover(null)}
        onClick={(e) => {
          const p = pick(e);
          if (p) onPickScalar(meta.name, p.r, p.c);
        }}
      />
      <div className="row-between muted-xs mono" style={{ marginTop: 6 }}>
        <span>
          {hover
            ? `(${hover.r}${is1D ? '' : `, ${hover.c}`}) = ${hover.v.toFixed(5)}`
            : 'Click any cell to zoom into that Single Scalar'}
        </span>
        <span>
          [{formatNum(meta.stats.min, 3)}, {formatNum(meta.stats.max, 3)}]
        </span>
      </div>
    </div>
  );
}

function InlineScalarView({
  client,
  manifest,
  replay,
  paramName,
  paramRow,
  paramCol,
  onPickScalar,
}: {
  client: TransformerEngineClient;
  manifest: ModelManifest;
  replay: TrainingReplayData | null;
  paramName: string;
  paramRow: number;
  paramCol: number;
  onPickScalar: (param: string, row: number, col: number) => void;
}) {
  const [val, setVal] = useState<number | null>(null);
  const meta =
    manifest.tensors.find((t) => t.name === paramName) ?? manifest.tensors[0];
  const is1D = meta.shape.length === 1;
  const r = Math.min(meta.shape[0] - 1, Math.max(0, paramRow));
  const c = is1D ? 0 : Math.min((meta.shape[1] ?? 1) - 1, Math.max(0, paramCol));

  useEffect(() => {
    let alive = true;
    client.getTensor(meta.name).then((arr) => {
      if (!alive) return;
      const idx = is1D ? r : r * (meta.shape[1] ?? 1) + c;
      setVal(arr[idx] ?? 0);
    });
    return () => {
      alive = false;
    };
  }, [client, meta, r, c, is1D]);

  const coordKey = is1D ? `${meta.name}[${r}]` : `${meta.name}[${r}, ${c}]`;
  const trajectory = useMemo(() => {
    if (!replay) return null;
    const pts = replay.checkpoints
      .map((ck) => ({ step: ck.step, val: ck.scalars[coordKey] }))
      .filter((p): p is { step: number; val: number } => typeof p.val === 'number');
    return pts.length > 1 ? pts : null;
  }, [replay, coordKey]);

  const maxHist = Math.max(...meta.histogram.counts, 1);

  return (
    <div className="lab-card">
      <div className="muted-xs mono">{coordKey}</div>
      <div className="scalar-readout mono">
        {val !== null ? val.toFixed(6) : '...'}
      </div>
      <div className="muted-xs" style={{ marginBottom: 8 }}>
        {meta.role}
      </div>

      <div className="hist-wrap" aria-label="Weight distribution histogram">
        {meta.histogram.counts.map((cnt, i) => {
          const e0 = meta.histogram.edges[i];
          const e1 = meta.histogram.edges[i + 1];
          const hit = val !== null && val >= e0 && val <= e1;
          return (
            <div
              key={i}
              className={`hist-col ${hit ? 'is-hit' : ''}`}
              style={{ height: `${Math.max(5, (cnt / maxHist) * 100)}%` }}
            />
          );
        })}
      </div>

      {trajectory ? (
        <MiniLineChart label="Recorded Step 0 → Final AdamW trajectory" points="{trajectory}"/>
      ) : (
        <div className="muted-xs" style={{ marginTop: 8 }}>
          Jump to one of the 12 checkpoint-tracked weights to plot its training trajectory:
        </div>
      )}

      {replay && (
        <div className="chip-wrap" style={{ marginTop: 8 }}>
          {replay.tracked_scalars_meta.slice(0, 6).map((m) => (
            <button
              key={m.key}
              type="button"
              className={`mini-btn mono ${m.key === coordKey ? 'is-active' : ''}`}
              onClick={() => onPickScalar(m.param, m.coord[0] ?? 0, m.coord[1] ?? 0)}
            >
              {m.key}
            </button>
          ))}
        </div>
      )}
    </div>
  );
}

function MiniLineChart({
  points,
  label,
}: {
  points: { step: number; val: number }[];
  label: string;
}) {
  const w = 340;
  const h = 86;
  const pad = 14;
  const maxS = Math.max(...points.map((p) => p.step), 1);
  const minV = Math.min(...points.map((p) => p.val));
  const maxV = Math.max(...points.map((p) => p.val));
  const span = Math.max(maxV - minV, 1e-4);
  const d = points
    .map((p, i) => {
      const x = pad + (p.step / maxS) * (w - pad * 2);
      const y = h - pad - ((p.val - minV) / span) * (h - pad * 2);
      return `${i === 0 ? 'M' : 'L'} ${x.toFixed(1)} ${y.toFixed(1)}`;
    })
    .join(' ');

  return (
    <div style={{ marginTop: 10 }}>
      <div className="muted-xs" style={{ marginBottom: 4 }}>
        {label}
      </div>
      <svg viewBox={`0 0 ${w} ${h}`} className="mini-svg">
        <path d={d} fill="none" stroke="#1d4ed8" strokeWidth="2.2" />
      </svg>
    </div>
  );
}

function InlineAblationLab({
  baseline,
  ablated,
  ablation,
  tokenizer,
  onChange,
}: {
  baseline: EngineTraceResult;
  ablated: EngineTraceResult | null;
  ablation: AblationConfig;
  tokenizer: TokenizerData;
  onChange: (next: AblationConfig) => void;
}) {
  const active = hasActiveAblation(ablation);

  const toggleHead = (l: number, h: number) => {
    const cur = ablation.disabledHeads ?? [];
    const exists = cur.some(([a, b]) => a === l && b === h);
    onChange({
      ...ablation,
      disabledHeads: exists
        ? cur.filter(([a, b]) => !(a === l && b === h))
        : [...cur, [l, h]],
    });
  };

  const toggleList = (
    key: 'disabledAttnBlocks' | 'disabledMlpBlocks' | 'disabledLayers',
    l: number
  ) => {
    const cur = ablation[key] ?? [];
    onChange({
      ...ablation,
      [key]: cur.includes(l) ? cur.filter((x) => x !== l) : [...cur, l],
    });
  };

  const toggleResidual = (kind: 'attn' | 'mlp', l: number) => {
    const cur = ablation.disabledResiduals ?? [];
    const exists = cur.some(([k, i]) => k === kind && i === l);
    onChange({
      ...ablation,
      disabledResiduals: exists
        ? cur.filter(([k, i]) => !(k === kind && i === l))
        : [...cur, [kind, l]],
    });
  };

  const toggleLn = (kind: 'ln_1' | 'ln_2' | 'ln_f', l: number) => {
    const cur = ablation.disabledLayerNorms ?? [];
    const exists = cur.some(([k, i]) => k === kind && i === l);
    onChange({
      ...ablation,
      disabledLayerNorms: exists
        ? cur.filter(([k, i]) => !(k === kind && i === l))
        : [...cur, [kind, l]],
    });
  };

  const rows = useMemo(() => {
    const ids = new Set<number>();
    baseline.trace.finalProbs
      .map((p, id) => ({ id, p }))
      .sort((a, b) => b.p - a.p)
      .slice(0, 5)
      .forEach((x) => ids.add(x.id));
    if (ablated) {
      ablated.trace.finalProbs
        .map((p, id) => ({ id, p }))
        .sort((a, b) => b.p - a.p)
        .slice(0, 3)
        .forEach((x) => ids.add(x.id));
    }
    return Array.from(ids)
      .map((id) => ({
        id,
        text: tokenizer.vocab[id],
        baseP: baseline.trace.finalProbs[id],
        ablP: ablated ? ablated.trace.finalProbs[id] : baseline.trace.finalProbs[id],
      }))
      .sort((a, b) => Math.max(b.baseP, b.ablP) - Math.max(a.baseP, a.ablP))
      .slice(0, 6);
  }, [baseline, ablated, tokenizer]);

  return (
    <div className="lab-card">
      <div className="row-between" style={{ marginBottom: 8 }}>
        <span className="lab-card-title" style={{ margin: 0 }}>
          Disable Live Components
        </span>
        {active && (
          <button type="button" className="mini-btn is-warm" onClick={() => onChange({})}>
            Reset all
          </button>
        )}
      </div>

      {[0, 1, 2, 3].map((l) => (
        <div key={l} className="ablate-line">
          <span className="mono ablate-tag">B{l}</span>
          <div className="chip-wrap">
            {[0, 1, 2, 3].map((h) => {
              const off = (ablation.disabledHeads ?? []).some(
                ([a, b]) => a === l && b === h
              );
              return (
                <button
                  key={h}
                  type="button"
                  className={`mini-btn mono ${off ? 'is-warm' : ''}`}
                  onClick={() => toggleHead(l, h)}
                >
                  H{h}
                </button>
              );
            })}
            <button
              type="button"
              className={`mini-btn ${(ablation.disabledAttnBlocks ?? []).includes(l) ? 'is-warm' : ''}`}
              onClick={() => toggleList('disabledAttnBlocks', l)}
            >
              Attn
            </button>
            <button
              type="button"
              className={`mini-btn ${(ablation.disabledMlpBlocks ?? []).includes(l) ? 'is-warm' : ''}`}
              onClick={() => toggleList('disabledMlpBlocks', l)}
            >
              MLP
            </button>
            <button
              type="button"
              className={`mini-btn ${
                (ablation.disabledResiduals ?? []).some(([k, i]) => k === 'attn' && i === l)
                  ? 'is-warm'
                  : ''
              }`}
              onClick={() => toggleResidual('attn', l)}
            >
              Res
            </button>
            <button
              type="button"
              className={`mini-btn ${
                (ablation.disabledLayerNorms ?? []).some(([k, i]) => k === 'ln_1' && i === l)
                  ? 'is-warm'
                  : ''
              }`}
              onClick={() => toggleLn('ln_1', l)}
            >
              LN1
            </button>
            <button
              type="button"
              className={`mini-btn ${(ablation.disabledLayers ?? []).includes(l) ? 'is-warm' : ''}`}
              onClick={() => toggleList('disabledLayers', l)}
            >
              Skip
            </button>
          </div>
        </div>
      ))}

      <div style={{ marginTop: 10 }}>
        {active && ablated && baseline.trace.inputLoss !== null && ablated.trace.inputLoss !== null && (
          <div className="muted-xs mono" style={{ marginBottom: 6 }}>
            Loss shift: {baseline.trace.inputLoss.toFixed(3)} → {ablated.trace.inputLoss.toFixed(3)} (
            {ablated.trace.inputLoss - baseline.trace.inputLoss >= 0 ? '+' : ''}
            {(ablated.trace.inputLoss - baseline.trace.inputLoss).toFixed(3)})
          </div>
        )}
        {rows.map((item) => (
          <div key={item.id} className="lens-row mono">
            <span>{formatTokenExplicit(item.text)}</span>
            <div className="dual-bars">
              <div className="bar-track">
                <div className="bar-fill" style={{ width: `${Math.min(100, item.baseP * 100)}%` }} />
              </div>
              {active && (
                <div className="bar-track">
                  <div className="bar-fill-warm" style={{ width: `${Math.min(100, item.ablP * 100)}%` }} />
                </div>
              )}
            </div>
            <span>
              {(item.baseP * 100).toFixed(0)}%{active ? `→${(item.ablP * 100).toFixed(0)}%` : ''}
            </span>
          </div>
        ))}
      </div>
    </div>
  );
}
```

```tsx
import { useState } from 'react';
import {
  ANATOMY_NODES,
  getAnatomyNode,
  getBreadcrumbChain,
  registerDisplayName,
} from '../content/llmAnatomy';
import type { Register } from '../state/atlasState';
import { NodeLab, type NodeLabProps } from './NodeLab';

const REGISTERS: Register[] = ['simple', 'sufficient', 'complete'];

interface AnatomyInspectorProps extends Omit<NodeLabProps, 'node'> {
  nodeId: string;
  register: Register;
  telemetryLines: string[];
  onSelectNode: (id: string) => void;
  onChangeRegister: (r: Register) => void;
}

export function AnatomyInspector(props: AnatomyInspectorProps) {
  const [showOutline, setShowOutline] = useState(false);
  const node = getAnatomyNode(props.nodeId);
  const breadcrumbs = getBreadcrumbChain(props.nodeId);
  const childNodes = node.children
    .map((id) => ANATOMY_NODES[id])
    .filter((n): n is NonNullable<typeof n> => Boolean(n));

  return (
    <aside className="inspector-rail" aria-live="polite" aria-label="LLM Anatomy Inspector">
      {/* Sticky Breadcrumb + 3-Mode Explanation Switch */}
      <div className="inspector-top">
        <div className="crumb-bar">
          <nav className="crumb-trail" aria-label="Anatomy path">
            {breadcrumbs.map((b, i) => (
              <span key={b.id} className="crumb-item">
                {i > 0 && <span className="crumb-sep">/</span>}
                <button
                  type="button"
                  className={`crumb-link ${b.id === node.id ? 'is-current' : ''}`}
                  onClick={() => {
                    setShowOutline(false);
                    props.onSelectNode(b.id);
                  }}
                >
                  {b.shortTitle}
                </button>
              </span>
            ))}
          </nav>

          <div className="crumb-actions">
            {node.parentId && (
              <button
                type="button"
                className="mini-btn"
                onClick={() => {
                  setShowOutline(false);
                  props.onSelectNode(node.parentId!);
                }}
              >
                ↑ Up
              </button>
            )}
            <button
              type="button"
              className={`mini-btn ${showOutline ? 'is-active' : ''}`}
              onClick={() => setShowOutline((v) => !v)}
            >
              {showOutline ? 'Close Map' : 'Full Map'}
            </button>
          </div>
        </div>

        <div className="register-switch" role="group" aria-label="Explanation depth">
          {REGISTERS.map((r) => (
            <button
              key={r}
              type="button"
              className={`reg-btn ${props.register === r ? 'is-active' : ''}`}
              onClick={() => props.onChangeRegister(r)}
            >
              {registerDisplayName(r)}
            </button>
          ))}
        </div>
      </div>

      {/* Body: Either Full Recursive Tree Map OR Current Node Deep Dive */}
      <div className="inspector-scroll">
        {showOutline ? (
          <div className="outline-view">
            <div className="section-label">Complete LLM Anatomy (34 Components)</div>
            <TreeBranch activeId="{props.nodeId}" depth="{0}" nodeId="root" onPick="{(id)"> {
                props.onSelectNode(id);
                setShowOutline(false);
              }}
            />
          </div>
        ) : (
          <>
            <div className="node-header">
              <div className="node-meta-row">
                <span
                  className={`status-pill ${
                    node.status === 'live' ? 'is-live' : 'is-ext'
                  }`}
                >
                  {node.status === 'live' ? 'Live in Miniature Model' : 'Modern Frontier Extension'}
                </span>
                {node.shapeTag && (
                  <span className="shape-pill mono">{node.shapeTag}</span>
                )}
              </div>
              <h1 className="node-title">{node.title}</h1>
            </div>

            <p className="node-prose">{node.copy[props.register]}</p>

            {props.register === 'simple' && node.simplificationNote && (
              <div className="note-banner">{node.simplificationNote}</div>
            )}

            {props.telemetryLines.length > 0 && (
              <div className="telemetry-strip mono">
                {props.telemetryLines.map((line, i) => (
                  <div key={i}>{line}</div>
                ))}
              </div>
            )}

            <NodeLab ablated="{props.ablated}" ablation="{props.ablation}" baseline="{props.baseline}" busy="{props.busy}" client="{props.client}" head="{props.head}" hoveredTokenIdx="{props.hoveredTokenIdx}" layer="{props.layer}" manifest="{props.manifest}" node="{node}" onChangeAblation="{props.onChangeAblation}" onChangeSampling="{props.onChangeSampling}" onHoverToken="{props.onHoverToken}" onInspectScalar="{props.onInspectScalar}" onStepAutoregressive="{props.onStepAutoregressive}" onTogglePreSoftmax="{props.onTogglePreSoftmax}" paramCol="{props.paramCol}" paramName="{props.paramName}" paramRow="{props.paramRow}" replay="{props.replay}" showPreSoftmax="{props.showPreSoftmax}" temperature="{props.temperature}" tokenizer="{props.tokenizer}" topK="{props.topK}" topP="{props.topP}"/>

            {childNodes.length > 0 && (
              <div className="subparts-section">
                <div className="section-label">
                  Inside {node.shortTitle} ({childNodes.length} subparts — click to expand)
                </div>
                <div className="subparts-list">
                  {childNodes.map((child) => (
                    <button
                      key={child.id}
                      type="button"
                      className="subpart-row"
                      onClick={() => props.onSelectNode(child.id)}
                    >
                      <div className="subpart-top">
                        <span className="subpart-name">{child.title}</span>
                        <span className="subpart-right">
                          {child.children.length > 0 && (
                            <span className="subpart-count mono">
                              {child.children.length} inside
                            </span>
                          )}
                          <span
                            className={`dot-indicator ${
                              child.status === 'live' ? 'is-live' : 'is-ext'
                            }`}
                            title={
                              child.status === 'live'
                                ? 'Live in model'
                                : 'Frontier extension'
                            }
                          />
                          <span className="subpart-arrow">›</span>
                        </span>
                      </div>
                      <div className="subpart-desc">{child.oneLiner}</div>
                    </button>
                  ))}
                </div>
              </div>
            )}
          </>
        )}
      </div>
    </aside>
  );
}

function TreeBranch({
  nodeId,
  activeId,
  depth,
  onPick,
}: {
  nodeId: string;
  activeId: string;
  depth: number;
  onPick: (id: string) => void;
}) {
  const node = ANATOMY_NODES[nodeId];
  if (!node) return null;

  return (
    <div>
      <button
        type="button"
        className={`tree-row ${node.id === activeId ? 'is-current' : ''}`}
        style={{ paddingLeft: 10 + depth * 14 }}
        onClick={() => onPick(node.id)}
      >
        <span className="tree-name">{node.shortTitle}</span>
        <span
          className={`dot-indicator ${
            node.status === 'live' ? 'is-live' : 'is-ext'
          }`}
        />
      </button>
      {node.children.map((cid) => (
        <TreeBranch + 1} activeId="{activeId}" depth="{depth" key="{cid}" nodeId="{cid}" onPick="{onPick}"/>
      ))}
    </div>
  );
}
```

```tsx
import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { AnatomyInspector } from './components/AnatomyInspector';
import type { TrainingReplayData } from './components/NodeLab';
import { getAnatomyNode, PRESET_PROMPTS } from './content/llmAnatomy';
import { TransformerEngineClient } from './engine/client';
import type {
  AblationConfig,
  ActiveLayerTrace,
  EngineInitResult,
  EngineTraceResult,
} from './engine/types';
import { ExplodedWorld } from './scene/ExplodedWorld';
import {
  hasActiveAblation,
  isHeadDisabled,
  stateFromUrl,
  stateToSearch,
  type AtlasState,
  type Register,
} from './state/atlasState';
import {
  computeSampledDistribution,
  countActiveNeurons,
  formatNum,
  formatToken,
  formatTokenExplicit,
  sampleFromDistribution,
  vectorL2Norm,
} from './utils/math';

export function App() {
  const mountRef = useRef<HTMLDivElement | null>(null);
  const worldRef = useRef<ExplodedWorld | null>(null);
  const clientRef = useRef<TransformerEngineClient | null>(null);

  const [state, setState] = useState<AtlasState>(() =>
    stateFromUrl(window.location.search)
  );
  const [draftPrompt, setDraftPrompt] = useState<string>(state.prompt);
  const [initInfo, setInitInfo] = useState<EngineInitResult | null>(null);
  const [replay, setReplay] = useState<TrainingReplayData | null>(null);
  const [baseline, setBaseline] = useState<EngineTraceResult | null>(null);
  const [ablated, setAblated] = useState<EngineTraceResult | null>(null);
  const [hoveredTokenIdx, setHoveredTokenIdx] = useState<number | null>(null);
  const [temperature, setTemperature] = useState<number>(0.8);
  const [topK, setTopK] = useState<number>(12);
  const [topP, setTopP] = useState<number>(0.9);
  const [copied, setCopied] = useState<boolean>(false);
  const [busy, setBusy] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);

  // Persist exact exploration state in URL
  useEffect(() => {
    const nextSearch = stateToSearch(state);
    if (window.location.search !== nextSearch) {
      window.history.replaceState(
        null,
        '',
        `${window.location.pathname}${nextSearch}`
      );
    }
  }, [state]);

  // Mount Three.js Exploded World once
  useEffect(() => {
    const el = mountRef.current;
    if (!el) return;

    const world = new ExplodedWorld(el, {
      onSelectNode: (nodeId, layer) => {
        setState((prev) => ({
          ...prev,
          nodeId,
          layer: typeof layer === 'number' ? layer : prev.layer,
        }));
      },
      onHoverToken: (idx) => {
        setHoveredTokenIdx(idx);
      },
    });
    worldRef.current = world;

    return () => {
      world.dispose();
      worldRef.current = null;
    };
  }, []);

  // Initialize verified Web Worker + load real weights & training replay
  useEffect(() => {
    const client = new TransformerEngineClient();
    clientRef.current = client;
    const modelBaseUrl = new URL(
      `${import.meta.env.BASE_URL}model/`,
      window.location.origin
    ).href;

    client
      .init(modelBaseUrl)
      .then((info) => setInitInfo(info))
      .catch((err) =>
        setError(err instanceof Error ? err.message : String(err))
      );

    fetch(`${modelBaseUrl}training_replay.json`)
      .then((res) => (res.ok ? res.json() : null))
      .then((data: TrainingReplayData | null) => {
        if (data) setReplay(data);
      })
      .catch(() => undefined);

    return () => {
      client.dispose();
      clientRef.current = null;
    };
  }, []);

  // Recompute real forward pass & ablation counterfactual when prompt or ablation changes
  useEffect(() => {
    const client = clientRef.current;
    if (!client || !initInfo) return;
    let alive = true;
    setBusy(true);
    setError(null);

    const run = async () => {
      try {
        const baseRes = await client.trace(state.prompt);
        if (!alive) return;
        setBaseline(baseRes);

        if (hasActiveAblation(state.ablation)) {
          const ablRes = await client.trace(state.prompt, state.ablation);
          if (!alive) return;
          setAblated(ablRes);
        } else {
          setAblated(null);
        }
      } catch (err) {
        if (alive) {
          setError(err instanceof Error ? err.message : String(err));
        }
      } finally {
        if (alive) setBusy(false);
      }
    };

    void run();
    return () => {
      alive = false;
    };
  }, [initInfo, state.prompt, state.ablation]);

  // Keep 3D exploded scene synced with active node & live trace
  useEffect(() => {
    if (!worldRef.current || !initInfo || !baseline) return;
    worldRef.current.update({
      nodeId: state.nodeId,
      layer: state.layer,
      head: state.head,
      showPreSoftmax: state.showPreSoftmax,
      ablation: state.ablation,
      baseline,
      ablated,
      tokenizer: initInfo.tokenizer,
      hoveredTokenIdx,
      temperature,
      topK,
      topP,
    });
  }, [
    initInfo,
    baseline,
    ablated,
    state.nodeId,
    state.layer,
    state.head,
    state.showPreSoftmax,
    state.ablation,
    hoveredTokenIdx,
    temperature,
    topK,
    topP,
  ]);

  // Keyboard shortcuts: Esc zooms up one parent folder, 1/2/3 switches register
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      const tag = (e.target as HTMLElement | null)?.tagName;
      if (tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT') return;
      if (e.key === 'Escape') {
        setState((prev) => {
          const cur = getAnatomyNode(prev.nodeId);
          return cur.parentId ? { ...prev, nodeId: cur.parentId } : prev;
        });
      } else if (e.key === '1') {
        setState((prev) => ({ ...prev, register: 'simple' }));
      } else if (e.key === '2') {
        setState((prev) => ({ ...prev, register: 'sufficient' }));
      } else if (e.key === '3') {
        setState((prev) => ({ ...prev, register: 'complete' }));
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);

  const handleSelectNode = useCallback((nodeId: string) => {
    setState((prev) => ({ ...prev, nodeId }));
  }, []);

  const handleChangeRegister = useCallback((register: Register) => {
    setState((prev) => ({ ...prev, register }));
  }, []);

  const handleChangeAblation = useCallback((ablation: AblationConfig) => {
    setState((prev) => ({ ...prev, ablation }));
  }, []);

  const handleInspectScalar = useCallback(
    (paramName: string, paramRow: number, paramCol: number) => {
      setState((prev) => ({
        ...prev,
        nodeId: 'scalar',
        paramName,
        paramRow,
        paramCol,
      }));
    },
    []
  );

  const handleTogglePreSoftmax = useCallback(() => {
    setState((prev) => ({
      ...prev,
      showPreSoftmax: !prev.showPreSoftmax,
    }));
  }, []);

  const handleChangeSampling = useCallback(
    (patch: { temperature?: number; topK?: number; topP?: number }) => {
      if (patch.temperature !== undefined) setTemperature(patch.temperature);
      if (patch.topK !== undefined) setTopK(patch.topK);
      if (patch.topP !== undefined) setTopP(patch.topP);
    },
    []
  );

  const handleStepAutoregressive = useCallback(() => {
    const active = ablated ?? baseline;
    if (!active || !initInfo) return;
    const dist = computeSampledDistribution(
      active.trace.finalLogits,
      temperature,
      topK,
      topP
    );
    const nextId = sampleFromDistribution(dist);
    const piece = initInfo.tokenizer.vocab[nextId] ?? '';
    if (!piece || piece === '<|eos|>') return;
    const nextPrompt = `${state.prompt}${piece}`;
    setDraftPrompt(nextPrompt);
    setState((prev) => ({ ...prev, prompt: nextPrompt }));
  }, [ablated, baseline, initInfo, temperature, topK, topP, state.prompt]);

  const activeNode = getAnatomyNode(state.nodeId);
  const activeTrace = ablated ?? baseline;
  const ablationOn = hasActiveAblation(state.ablation);

  const telemetryLines = useMemo(() => {
    if (!activeTrace || !initInfo) return [];
    const { trace, tokens } = activeTrace;
    const T = tokens.length;
    const last = T - 1;
    const topId = trace.greedyNextTokenId;
    const topTok = formatTokenExplicit(initInfo.tokenizer.vocab[topId] ?? '');
    const topProb = (trace.finalProbs[topId] * 100).toFixed(1);

    if (activeNode.sceneView === 'embed') {
      return [
        `Tokens: ${T} / ${initInfo.manifest.config.max_seq_len} · Last token: "${formatTokenExplicit(tokens[last].text)}" (#${tokens[last].id})`,
        `||wte|| = ${formatNum(vectorL2Norm(trace.tokEmb[last]), 3)} · ||wpe[${last}]|| = ${formatNum(vectorL2Norm(trace.posEmb[last]), 3)}`,
      ];
    }

    const lt = trace.layers[state.layer];
    if (activeNode.sceneView === 'block') {
      if (lt.skipped) {
        return [`Block ${state.layer} is skipped by Ablation Lab (identity pass-through).`];
      }
      const active: ActiveLayerTrace = lt;
      return [
        `Block ${state.layer} · ||h_in||=${formatNum(vectorL2Norm(active.residualBeforeAttn[last]), 2)} → +Attn(${formatNum(vectorL2Norm(active.attn.output[last]), 2)}) → +MLP(${formatNum(vectorL2Norm(active.mlp.output[last]), 2)})`,
        `||h_out|| = ${formatNum(vectorL2Norm(active.residualAfterMlp[last]), 2)} · shape [${T} × 128]`,
      ];
    }

    if (activeNode.sceneView === 'attention') {
      if (lt.skipped) {
        return [`Block ${state.layer} is skipped by Ablation Lab.`];
      }
      const active: ActiveLayerTrace = lt;
      const focusIdx = hoveredTokenIdx ?? last;
      const row = active.attn.attnWeights[state.head][focusIdx];
      let bestCol = 0;
      for (let c = 1; c <= focusIdx; c++) {
        if (row[c] > row[bestCol]) bestCol = c;
      }
      const sum = row.reduce((a, b) => a + b, 0);
      const off = isHeadDisabled(state.ablation, state.layer, state.head);
      return [
        `Block ${state.layer} Head ${state.head}${off ? ' (OFF)' : ''} · "${formatTokenExplicit(tokens[focusIdx].text)}" attends most to "${formatTokenExplicit(tokens[bestCol].text)}" (${(row[bestCol] * 100).toFixed(1)}%)`,
        `Row softmax sum = ${sum.toFixed(6)} · Q, K, V shape [${T} × 32]`,
      ];
    }

    if (activeNode.sceneView === 'mlp') {
      if (lt.skipped) {
        return [`Block ${state.layer} is skipped by Ablation Lab.`];
      }
      const active: ActiveLayerTrace = lt;
      const firing = countActiveNeurons(active.mlp.hidden[last], 0.05);
      return [
        `Block ${state.layer} MLP · Token "${formatTokenExplicit(tokens[last].text)}": ${firing} / 512 GELU neurons active (>0.05)`,
        `||Δmlp|| = ${formatNum(vectorL2Norm(active.mlp.output[last]), 2)} · expansion 128 → 512 → 128`,
      ];
    }

    if (activeNode.sceneView === 'output') {
      const pool = computeSampledDistribution(
        trace.finalLogits,
        temperature,
        topK,
        topP
      );
      return [
        `Top logit: "${topTok}" z=${formatNum(trace.finalLogits[topId], 2)} → P=${topProb}%`,
        `Sampling nucleus: ${pool.length} tokens (τ=${temperature.toFixed(2)}, k=${topK}, p=${topP.toFixed(2)})`,
      ];
    }

    return [
      `${initInfo.manifest.total_params.toLocaleString()} real params · ${T} tokens · traced in ${activeTrace.elapsedMs}ms`,
      `Next prediction: "${topTok}" (${topProb}%)${trace.inputLoss !== null ? ` · Prompt loss: ${trace.inputLoss.toFixed(3)}` : ''}`,
    ];
  }, [
    activeTrace,
    initInfo,
    activeNode.sceneView,
    state.layer,
    state.head,
    state.ablation,
    hoveredTokenIdx,
    temperature,
    topK,
    topP,
  ]);

  const copyLink = async () => {
    try {
      await navigator.clipboard.writeText(window.location.href);
      setCopied(true);
      setTimeout(() => setCopied(false), 1600);
    } catch {
      setCopied(false);
    }
  };

  const showBlockSelector =
    activeNode.sceneView === 'block' ||
    activeNode.sceneView === 'attention' ||
    activeNode.sceneView === 'mlp';

  return (
    <div className="app-shell">
      <header className="topbar">
        <button
          type="button"
          className="brand-btn"
          onClick={() => handleSelectNode('root')}
          title="Zoom out to the complete LLM"
        >
          LLM Atlas
        </button>

        <form
          className="prompt-form"
          onSubmit={(e) => {
            e.preventDefault();
            if (!draftPrompt.trim()) return;
            setState((prev) => ({ ...prev, prompt: draftPrompt }));
          }}
        >
          <input
            className="prompt-input"
            value={draftPrompt}
            onChange={(e) => setDraftPrompt(e.target.value)}
            placeholder="Type any sentence to trace live..."
            aria-label="Prompt sentence"
          />
          <select
            className="preset-select"
            value=""
            aria-label="Example prompts"
            onChange={(e) => {
              const p = PRESET_PROMPTS.find((x) => x.id === e.target.value);
              if (!p) return;
              setDraftPrompt(p.text);
              setState((prev) => ({ ...prev, prompt: p.text }));
            }}
          >
            <option value="">Examples...</option>
            {PRESET_PROMPTS.map((p) => (
              <option key={p.id} value={p.id}>
                {p.label}
              </option>
            ))}
          </select>
          <button type="submit" className="primary-btn" disabled={busy || !initInfo}>
            {busy ? 'Tracing...' : 'Run'}
          </button>
        </form>

        <div className="topbar-actions">
          <button
            type="button"
            className={`mini-btn ${ablationOn ? 'is-warm' : state.nodeId === 'ablation' ? 'is-active' : ''}`}
            onClick={() => handleSelectNode('ablation')}
          >
            {ablationOn ? 'Ablation (on)' : 'Ablate'}
          </button>
          <button
            type="button"
            className="mini-btn"
            onClick={() => void copyLink()}
          >
            {copied ? 'Copied' : 'Share'}
          </button>
        </div>
      </header>

      <div className="split-workspace">
        <section className="stage-pane" aria-label="Interactive 3D exploded LLM">
          <div ref={mountRef} className="canvas-mount" />

          {!initInfo && !error && (
            <div className="loading-overlay">
              Loading 862,464 real trained parameters...
            </div>
          )}
          {error && <div className="error-banner">{error}</div>}

          <div className="stage-hud-top">
            <div className="stage-pill-group">
              {state.nodeId !== 'root' && (
                <button
                  type="button"
                  className="mini-btn"
                  onClick={() => handleSelectNode('root')}
                >
                  Zoom Out (Full LLM)
                </button>
              )}

              {showBlockSelector && (
                <>
                  <span className="stage-pill-label">Block</span>
                  {[0, 1, 2, 3].map((l) => (
                    <button
                      key={l}
                      type="button"
                      className={`mini-btn mono ${state.layer === l ? 'is-active' : ''}`}
                      onClick={() => setState((prev) => ({ ...prev, layer: l }))}
                    >
                      {l}
                    </button>
                  ))}
                </>
              )}

              {activeNode.sceneView === 'attention' && (
                <>
                  <span className="stage-pill-label">Head</span>
                  {[0, 1, 2, 3].map((h) => {
                    const off = isHeadDisabled(state.ablation, state.layer, h);
                    return (
                      <button
                        key={h}
                        type="button"
                        className={`mini-btn mono ${
                          off ? 'is-warm' : state.head === h ? 'is-active' : ''
                        }`}
                        onClick={() => setState((prev) => ({ ...prev, head: h }))}
                      >
                        H{h}
                      </button>
                    );
                  })}
                </>
              )}
            </div>

            {ablationOn && (
              <div className="stage-pill-group">
                <span className="stage-pill-label">Ablation active</span>
                <button
                  type="button"
                  className="mini-btn is-warm"
                  onClick={() => handleChangeAblation({})}
                >
                  Reset
                </button>
              </div>
            )}
          </div>

          {activeTrace && (
            <div className="stage-hud-bottom">
              <div className="token-strip-float" aria-label="Prompt tokens">
                {activeTrace.tokens.map((tok, idx) => (
                  <button
                    key={`${tok.start}-${idx}`}
                    type="button"
                    className={`tok-badge mono ${
                      hoveredTokenIdx === idx ? 'is-hover' : ''
                    }`}
                    onMouseEnter={() => setHoveredTokenIdx(idx)}
                    onMouseLeave={() => setHoveredTokenIdx(null)}
                    title={`Slot ${idx} · Token #${tok.id}`}
                  >
                    {formatToken(tok.text)}
                  </button>
                ))}
              </div>
            </div>
          )}
        </section>

        {initInfo && baseline && clientRef.current ? (
          <AnatomyInspector
            nodeId={state.nodeId}
            register={state.register}
            telemetryLines={telemetryLines}
            client={clientRef.current}
            manifest={initInfo.manifest}
            tokenizer={initInfo.tokenizer}
            baseline={baseline}
            ablated={ablated}
            ablation={state.ablation}
            layer={state.layer}
            head={state.head}
            showPreSoftmax={state.showPreSoftmax}
            hoveredTokenIdx={hoveredTokenIdx}
            replay={replay}
            paramName={state.paramName}
            paramRow={state.paramRow}
            paramCol={state.paramCol}
            temperature={temperature}
            topK={topK}
            topP={topP}
            busy={busy}
            onSelectNode={handleSelectNode}
            onChangeRegister={handleChangeRegister}
            onChangeAblation={handleChangeAblation}
            onInspectScalar={handleInspectScalar}
            onTogglePreSoftmax={handleTogglePreSoftmax}
            onHoverToken={setHoveredTokenIdx}
            onChangeSampling={handleChangeSampling}
            onStepAutoregressive={handleStepAutoregressive}
          />
        ) : (
          <aside className="inspector-rail">
            <div className="inspector-scroll muted-sm">
              Initializing verified miniature transformer engine...
            </div>
          </aside>
        )}

    </div>
```
